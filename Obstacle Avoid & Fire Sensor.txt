// ======================================================
//       ESP32 FIRE + OBSTACLE AVOIDING ROBOT CAR
//                  FINAL 90° TURN VERSION
// ======================================================
//
// Board        : DOIT ESP32 DEVKIT V1
// Motor Driver : L298N
// Fire Sensor  : LM393 Flame Sensor
// Servo        : Scanning Servo
// Ultrasonic   : HC-SR04
//
// ======================================================

#include <ESP32Servo.h>


// ======================================================
// L298N
// ======================================================

#define ENA 14
#define IN1 27
#define IN2 26

#define ENB 13
#define IN3 25
#define IN4 33


// ======================================================
// FIRE SENSOR
// ======================================================

#define FIRE_SENSOR 34


// ======================================================
// SERVO
// ======================================================

#define SERVO_PIN 15


// ======================================================
// ULTRASONIC
// ======================================================

#define TRIG_PIN 5
#define ECHO_PIN 18


// ======================================================
// SERVO OBJECT
// ======================================================

Servo scanningServo;


// ======================================================
// SETTINGS
// ======================================================

const int OBSTACLE_DISTANCE = 25;


// ======================================================
// SERVO SETTINGS
// ======================================================
// তোমার আগের servo setup-ই রাখা হয়েছে

const int SERVO_CENTER = 90;
const int SERVO_RIGHT  = 150;
const int SERVO_LEFT   = 30;


// ======================================================
// BACKWARD SETTINGS
// ======================================================

const int BACKWARD_TIME = 500;


// ======================================================
// 90 DEGREE TURN SETTINGS
// ======================================================
//
// IMPORTANT:
//
// Robot একবারে continuous pivot করবে।
//
// আগের 700 ms-এ খুব অল্প turn হচ্ছিল,
// তাই সময় অনেক বাড়ানো হয়েছে।
//
// প্রথমে 4000 ms দিয়ে test করো।
//
// যদি 90° না হয়:
// কম হলে 4500 / 5000 করো
// বেশি হলে 3500 / 3000 করো
//
// ======================================================

const int TURN_90_TIME = 4000;


// ======================================================
// SERVO SETTLE TIME
// ======================================================

const int SERVO_SETTLE_TIME = 600;


// ======================================================
// SETUP
// ======================================================

void setup() {

  Serial.begin(115200);

  delay(500);


  // ====================================================
  // MOTOR PINS
  // ====================================================

  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);


  // ====================================================
  // FIRE SENSOR
  // ====================================================

  pinMode(FIRE_SENSOR, INPUT);


  // ====================================================
  // ULTRASONIC
  // ====================================================

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);


  // ====================================================
  // MOTOR OFF
  // ====================================================

  stopCar();


  // ====================================================
  // SERVO INITIALIZATION
  // ====================================================

  scanningServo.setPeriodHertz(50);

  scanningServo.attach(
    SERVO_PIN,
    500,
    2400
  );


  // ====================================================
  // SERVO CENTER
  // ====================================================

  Serial.println();
  Serial.println("==============================================");
  Serial.println("      ESP32 FIRE + OBSTACLE CAR");
  Serial.println("          90 DEGREE TURN VERSION");
  Serial.println("==============================================");

  Serial.println("System Power ON");
  Serial.println("Centering Servo...");


  scanningServo.write(SERVO_CENTER);

  delay(1000);


  Serial.println("Servo CENTERED at 90°");

  delay(500);


  Serial.println("CAR READY");
  Serial.println();
}


// ======================================================
// LOOP
// ======================================================

void loop() {


  // ====================================================
  // FIRE SENSOR
  // ====================================================

  int fireState = digitalRead(FIRE_SENSOR);


  // ====================================================
  // FIRE DETECTED
  // ====================================================

  if (fireState == LOW) {

    Serial.println();
    Serial.println("🔥 FIRE DETECTED!");
    Serial.println("🛑 CAR STOPPED");

    stopCar();

    delay(200);

    return;
  }


  // ====================================================
  // FRONT DISTANCE
  // ====================================================

  scanningServo.write(SERVO_CENTER);

  delay(50);

  int frontDistance = getDistance();


  Serial.print("Front Distance: ");
  Serial.print(frontDistance);
  Serial.println(" cm");


  // ====================================================
  // PATH CLEAR
  // ====================================================

  if (frontDistance > OBSTACLE_DISTANCE) {

    Serial.println("🚗 Path Clear");
    Serial.println("➡️ Moving Forward");

    moveForward();

    delay(100);

    return;
  }


  // ====================================================
  // OBSTACLE DETECTED
  // ====================================================

  Serial.println();
  Serial.println("=================================");
  Serial.println("⚠️ OBSTACLE DETECTED");
  Serial.println("🛑 STOPPING");
  Serial.println("=================================");


  stopCar();

  delay(300);


  // ====================================================
  // BACKWARD FIRST
  // ====================================================

  Serial.println("🔙 Moving BACKWARD");


  scanningServo.write(SERVO_CENTER);

  delay(200);


  moveBackward();

  delay(BACKWARD_TIME);


  stopCar();

  delay(300);


  // ====================================================
  // CHECK RIGHT
  // ====================================================

  Serial.println();
  Serial.println("➡️ Checking RIGHT");


  scanningServo.write(SERVO_RIGHT);

  delay(SERVO_SETTLE_TIME);


  int rightDistance = getDistance();


  Serial.print("RIGHT Distance: ");
  Serial.print(rightDistance);
  Serial.println(" cm");


  // ====================================================
  // RIGHT CLEAR
  // ====================================================

  if (rightDistance > OBSTACLE_DISTANCE) {

    Serial.println("✅ RIGHT CLEAR");
    Serial.println("↪️ STARTING 90° RIGHT TURN");


    // Servo center
    scanningServo.write(SERVO_CENTER);

    delay(300);


    // ==================================================
    // ONE CONTINUOUS 90° TURN
    // ==================================================

    turnRight90();


    Serial.println("🛑 RIGHT TURN COMPLETE");

    delay(300);

    return;
  }


  // ====================================================
  // RIGHT BLOCKED
  // ====================================================

  Serial.println("❌ RIGHT BLOCKED");


  // ====================================================
  // CHECK LEFT
  // ====================================================

  Serial.println();
  Serial.println("⬅️ Checking LEFT");


  scanningServo.write(SERVO_LEFT);

  delay(SERVO_SETTLE_TIME);


  int leftDistance = getDistance();


  Serial.print("LEFT Distance: ");
  Serial.print(leftDistance);
  Serial.println(" cm");


  // ====================================================
  // LEFT CLEAR
  // ====================================================

  if (leftDistance > OBSTACLE_DISTANCE) {

    Serial.println("✅ LEFT CLEAR");
    Serial.println("↩️ STARTING 90° LEFT TURN");


    // Servo center
    scanningServo.write(SERVO_CENTER);

    delay(300);


    // ==================================================
    // ONE CONTINUOUS 90° TURN
    // ==================================================

    turnLeft90();


    Serial.println("🛑 LEFT TURN COMPLETE");

    delay(300);

    return;
  }


  // ====================================================
  // BOTH BLOCKED
  // ====================================================

  Serial.println("❌ RIGHT BLOCKED");
  Serial.println("❌ LEFT BLOCKED");

  Serial.println("🚧 BOTH SIDES BLOCKED");


  // ====================================================
  // BACKWARD AGAIN
  // ====================================================

  Serial.println("🔙 Moving BACKWARD Again");


  scanningServo.write(SERVO_CENTER);

  delay(200);


  moveBackward();

  delay(900);


  stopCar();

  delay(300);


  Serial.println("🔄 Checking Again");
  Serial.println();
}


// ======================================================
// ULTRASONIC DISTANCE
// ======================================================

int getDistance() {

  digitalWrite(TRIG_PIN, LOW);

  delayMicroseconds(2);


  digitalWrite(TRIG_PIN, HIGH);

  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);


  long duration = pulseIn(
    ECHO_PIN,
    HIGH,
    30000
  );


  if (duration == 0) {

    return 400;
  }


  int distance = duration * 0.0343 / 2;


  if (distance <= 0) {

    return 400;
  }


  return distance;
}


// ======================================================
// MOVE FORWARD
// ======================================================

void moveForward() {

  // LEFT MOTOR -> FORWARD

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);


  // RIGHT MOTOR -> FORWARD

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);


  digitalWrite(ENA, HIGH);
  digitalWrite(ENB, HIGH);
}


// ======================================================
// MOVE BACKWARD
// ======================================================

void moveBackward() {

  // LEFT MOTOR -> BACKWARD

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);


  // RIGHT MOTOR -> BACKWARD

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);


  digitalWrite(ENA, HIGH);
  digitalWrite(ENB, HIGH);
}


// ======================================================
// STOP
// ======================================================

void stopCar() {

  digitalWrite(ENA, LOW);
  digitalWrite(ENB, LOW);


  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


// ======================================================
// RIGHT 90° TURN
// ======================================================
//
// LEFT MOTOR  -> FORWARD
// RIGHT MOTOR -> BACKWARD
//
// দুটো motor একই সময়ে বিপরীত দিকে ঘুরবে।
// ফলে robot নিজের জায়গার কাছাকাছি pivot করবে।
//
// এটি ছোট ছোট step নয়।
// একটি continuous movement।
// ======================================================

void turnRight90() {

  Serial.println("↪️ RIGHT 90° TURN START");


  // LEFT MOTOR FORWARD

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);


  // RIGHT MOTOR BACKWARD

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);


  // FULL POWER

  digitalWrite(ENA, HIGH);
  digitalWrite(ENB, HIGH);


  // ==================================================
  // CONTINUOUS TURN
  // ==================================================

  delay(TURN_90_TIME);


  // ==================================================
  // IMMEDIATE STOP
  // ==================================================

  stopCar();


  Serial.println("🛑 RIGHT 90° TURN STOP");
}


// ======================================================
// LEFT 90° TURN
// ======================================================
//
// LEFT MOTOR  -> BACKWARD
// RIGHT MOTOR -> FORWARD
//
// ======================================================

void turnLeft90() {

  Serial.println("↩️ LEFT 90° TURN START");


  // LEFT MOTOR BACKWARD

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);


  // RIGHT MOTOR FORWARD

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);


  // FULL POWER

  digitalWrite(ENA, HIGH);
  digitalWrite(ENB, HIGH);


  // ==================================================
  // CONTINUOUS TURN
  // ==================================================

  delay(TURN_90_TIME);


  // ==================================================
  // IMMEDIATE STOP
  // ==================================================

  stopCar();


  Serial.println("🛑 LEFT 90° TURN STOP");
}
